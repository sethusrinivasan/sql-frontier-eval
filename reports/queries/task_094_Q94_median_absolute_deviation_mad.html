4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted
Database Engine Context: This query was executed and profiled on DuckDB 2.0 In-Memory Engine with equivalent analytical syntax and execution verification on PostgreSQL 18.7 + Extensions (Docker Compose).
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 241.66 ms) 
WITH med AS
  (SELECT median(amount) AS overall_median
   FROM payment_volumes),
     deviations AS
  (SELECT p.tx_id,
          p.amount,
          m.overall_median,
          abs(p.amount - m.overall_median) AS abs_deviation
   FROM payment_volumes p
   CROSS JOIN med m),
     mad_val AS
  (SELECT median(abs_deviation) AS mad_score
   FROM deviations)
SELECT d.tx_id,
       d.amount,
       d.overall_median,
       round(d.abs_deviation / nullif(v.mad_score, 0), 2) AS robust_mad_score,
       CASE
           WHEN (d.abs_deviation / nullif(v.mad_score, 0)) > 3.0 THEN 'ROBUST_OUTLIER'
           ELSE 'NORMAL'
       END AS anomaly_flag
FROM deviations d
CROSS JOIN mad_val v
ORDER BY robust_mad_score DESC;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 239.83 ms) 
-- Compute Median Absolute Deviation (MAD) and flag robust outliers
 -- MAD = MEDIAN( |x - MEDIAN(x)| )
 WITH med AS
  (SELECT median(amount) AS overall_median
   FROM payment_volumes),
      deviations AS
  (SELECT p.tx_id,
          p.amount,
          m.overall_median,
          abs(p.amount - m.overall_median) AS abs_deviation
   FROM payment_volumes p
   CROSS JOIN med m),
      mad_val AS
  (SELECT median(abs_deviation) AS mad_score
   FROM deviations)
SELECT d.tx_id,
       d.amount,
       d.overall_median,
       round(d.abs_deviation / nullif(v.mad_score, 0), 2) AS robust_mad_score,
       CASE
           WHEN (d.abs_deviation / nullif(v.mad_score, 0)) > 3.0 THEN 'ROBUST_OUTLIER'
           ELSE 'NORMAL'
       END AS anomaly_flag
FROM deviations d
CROSS JOIN mad_val v
ORDER BY robust_mad_score DESC;
