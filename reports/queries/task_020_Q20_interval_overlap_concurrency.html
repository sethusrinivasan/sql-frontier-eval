4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 157.34 ms) 
WITH EVENTS AS
  (SELECT start_time AS event_time,
          1 AS change_val
   FROM staff_shifts
   UNION ALL SELECT end_time AS event_time,
                    -1 AS change_val
   FROM staff_shifts), timeline AS
  (-- If shifts can end exactly when another starts, handle ordering:
 -- In standard concurrency calculations, whether end is inclusive or exclusive matters.
 -- Usually, interval is [start_time, end_time). If a shift ends at 12:00 and another starts at 12:00,
 -- departing (-1) happens before arriving (+1) or at the exact instant.
 -- When ordering by event_time, ordering -1 before +1 treats intervals as half-open [start, end),
 -- or summing all changes at the exact same timestamp.
 -- Grouping by event_time or summing over window:
 SELECT event_time,
        sum(change_val) AS net_change
   FROM EVENTS
   GROUP BY event_time),
                       running_total AS
  (SELECT sum(net_change) OVER (
                                ORDER BY event_time ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS concurrent_workers
   FROM timeline)
SELECT coalesce(max(concurrent_workers), 0) AS peak_concurrent_workers
FROM running_total;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 113.51 ms) 
-- Transform start/end intervals into delta events and compute peak concurrent staffing
 WITH shift_events AS
  (SELECT start_time AS event_time,
          1 AS delta
   FROM staff_shifts
   UNION ALL SELECT end_time AS event_time,
                    -1 AS delta
   FROM staff_shifts),
      running_occupancy AS
  (SELECT event_time,
          sum(delta) OVER (
                           ORDER BY event_time, delta DESC) AS concurrent_workers
   FROM shift_events)
SELECT max(concurrent_workers) AS peak_concurrent_workers
FROM running_occupancy;
