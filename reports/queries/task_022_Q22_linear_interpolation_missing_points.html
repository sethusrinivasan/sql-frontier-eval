4. Queries Executed (Candidate vs Golden Ground Truth) #

Beautified & Syntax Highlighted
Database Engine Context: This query was executed and profiled on DuckDB 2.0 In-Memory Engine with equivalent analytical syntax and execution verification on PostgreSQL 18.7 + Extensions (Docker Compose) and Amazon Aurora Serverless v2 PostgreSQL.
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 78.88 ms) 
WITH bounds AS
  (SELECT ts,
          psi,
          last_value(psi
                     IGNORE nulls) OVER (
                                         ORDER BY ts ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS prev_psi,
          last_value(ts
                     IGNORE nulls) OVER (
                                         ORDER BY ts ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS prev_ts,
          first_value(psi
                      IGNORE nulls) OVER (
                                          ORDER BY ts ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING) AS next_psi,
          first_value(ts
                      IGNORE nulls) OVER (
                                          ORDER BY ts ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING) AS next_ts
   FROM pressure_readings)
SELECT ts,
       psi AS raw_psi,
       CASE
           WHEN psi IS NOT NULL THEN psi
           ELSE round(prev_psi + (next_psi - prev_psi) * (date_diff('second', prev_ts, ts) * 1.0 / date_diff('second', prev_ts, next_ts)), 2)
       END AS interpolated_psi
FROM bounds
ORDER BY ts;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 76.88 ms) 
-- Linearly interpolate missing PSI values based on preceding and following known points
 WITH bounds AS
  (SELECT ts,
          psi,
          last_value(psi
                     IGNORE nulls) OVER (
                                         ORDER BY ts ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS prev_psi,
          last_value(ts
                     IGNORE nulls) OVER (
                                         ORDER BY ts ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS prev_ts,
          first_value(psi
                      IGNORE nulls) OVER (
                                          ORDER BY ts ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING) AS next_psi,
          first_value(ts
                      IGNORE nulls) OVER (
                                          ORDER BY ts ROWS BETWEEN CURRENT ROW AND UNBOUNDED FOLLOWING) AS next_ts
   FROM pressure_readings)
SELECT ts,
       psi AS raw_psi,
       CASE
           WHEN psi IS NOT NULL THEN psi
           ELSE round(prev_psi + (next_psi - prev_psi) * (date_diff('second', prev_ts, ts) * 1.0 / date_diff('second', prev_ts, next_ts)), 2)
       END AS interpolated_psi
FROM bounds
ORDER BY ts;
