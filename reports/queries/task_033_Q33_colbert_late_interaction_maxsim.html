4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted
Database Engine Context: This query was executed and profiled on DuckDB 2.0 In-Memory Engine with equivalent analytical syntax and execution verification on PostgreSQL 18.7 + Extensions (Docker Compose).
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 211.54 ms) 
WITH token_sims AS
  (SELECT q.q_token_id,
          q.token_text AS q_token,
          d.doc_id,
          d.token_text AS d_token,
          array_cosine_similarity(q.vec, d.vec) AS sim
   FROM query_tokens q
   CROSS JOIN doc_token_embeddings d),
     max_sim_per_query_token AS
  (SELECT doc_id,
          q_token_id,
          max(sim) AS max_token_sim
   FROM token_sims
   GROUP BY doc_id,
            q_token_id)
SELECT doc_id,
       round(sum(max_token_sim), 4) AS colbert_maxsim_score
FROM max_sim_per_query_token
GROUP BY doc_id
ORDER BY colbert_maxsim_score DESC;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 186.73 ms) 
-- ColBERT MaxSim calculation: SUM_q( MAX_d( sim(q, d) ) )
 WITH token_sims AS
  (SELECT q.q_token_id,
          q.token_text AS q_token,
          d.doc_id,
          d.token_text AS d_token,
          array_cosine_similarity(q.vec, d.vec) AS sim
   FROM query_tokens q
   CROSS JOIN doc_token_embeddings d),
      max_sim_per_query_token AS
  (SELECT doc_id,
          q_token_id,
          max(sim) AS max_token_sim
   FROM token_sims
   GROUP BY doc_id,
            q_token_id)
SELECT doc_id,
       round(sum(max_token_sim), 4) AS colbert_maxsim_score
FROM max_sim_per_query_token
GROUP BY doc_id
ORDER BY colbert_maxsim_score DESC;
