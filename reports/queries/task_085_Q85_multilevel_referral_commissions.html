4. Queries Executed (Candidate vs Golden Ground Truth) #

Beautified & Syntax Highlighted
Database Engine Context: This query was executed and profiled on DuckDB 2.0 In-Memory Engine with equivalent analytical syntax and execution verification on PostgreSQL 18.7 + Extensions (Docker Compose) and Amazon Aurora Serverless v2 PostgreSQL.
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 2.14 ms) 
WITH RECURSIVE upline AS
  (-- Level 1: Direct referrer
 SELECT r.user_id AS buyer_id,
        r.referrer_id AS beneficiary_id,
        1 AS tier,
        0.10 AS commission_rate
   FROM referral_links r
   WHERE r.user_id = 4
   UNION ALL -- Level 2 and 3 upline
 SELECT u.buyer_id,
        r.referrer_id AS beneficiary_id,
        u.tier + 1,
        CASE
            WHEN u.tier + 1 = 2 THEN 0.05
            ELSE 0.025
        END AS commission_rate
   FROM upline u
   JOIN referral_links r ON u.beneficiary_id = r.user_id
   WHERE u.tier < 3)
SELECT u.buyer_id,
       u.beneficiary_id,
       u.tier,
       p.amount AS purchase_amount,
       round(p.amount * u.commission_rate, 2) AS commission_earned
FROM upline u
JOIN user_purchases p ON u.buyer_id = p.user_id
ORDER BY u.tier;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 2.26 ms) 
-- Calculate multi-tier referral payout for purchase 101
 WITH RECURSIVE upline AS
  (-- Level 1: Direct referrer
 SELECT r.user_id AS buyer_id,
        r.referrer_id AS beneficiary_id,
        1 AS tier,
        0.10 AS commission_rate
   FROM referral_links r
   WHERE r.user_id = 4
   UNION ALL -- Level 2 and 3 upline
 SELECT u.buyer_id,
        r.referrer_id AS beneficiary_id,
        u.tier + 1,
        CASE
            WHEN u.tier + 1 = 2 THEN 0.05
            ELSE 0.025
        END AS commission_rate
   FROM upline u
   JOIN referral_links r ON u.beneficiary_id = r.user_id
   WHERE u.tier < 3)
SELECT u.buyer_id,
       u.beneficiary_id,
       u.tier,
       p.amount AS purchase_amount,
       round(p.amount * u.commission_rate, 2) AS commission_earned
FROM upline u
JOIN user_purchases p ON u.buyer_id = p.user_id
ORDER BY u.tier;
