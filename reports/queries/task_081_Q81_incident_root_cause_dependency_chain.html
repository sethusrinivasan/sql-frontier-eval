4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 0.76 ms) 
WITH RECURSIVE upstream_chain AS
  (SELECT service,
          depends_on,
          1 AS depth
   FROM service_mesh
   WHERE service = 'checkout-api'
   UNION ALL SELECT m.service,
                    m.depends_on,
                    c.depth + 1
   FROM upstream_chain c
   JOIN service_mesh m ON c.depends_on = m.service)
SELECT c.depth,
       c.service,
       c.depends_on AS upstream_service,
       a.status AS upstream_status,
       a.error_rate AS upstream_error_rate
FROM upstream_chain c
JOIN service_alerts a ON c.depends_on = a.service
ORDER BY c.depth DESC
LIMIT 1;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 0.87 ms) 
-- Trace dependencies backwards from checkout-api to find root critical failure
 WITH RECURSIVE upstream_chain AS
  (SELECT service,
          depends_on,
          1 AS depth
   FROM service_mesh
   WHERE service = 'checkout-api'
   UNION ALL SELECT m.service,
                    m.depends_on,
                    c.depth + 1
   FROM upstream_chain c
   JOIN service_mesh m ON c.depends_on = m.service)
SELECT c.depth,
       c.service,
       c.depends_on AS upstream_service,
       a.status AS upstream_status,
       a.error_rate AS upstream_error_rate
FROM upstream_chain c
JOIN service_alerts a ON c.depends_on = a.service
ORDER BY c.depth DESC
LIMIT 1;
