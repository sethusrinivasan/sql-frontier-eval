4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 495.88 ms) 
SELECT a.sku_id AS sku_a,
       a.merchant AS merchant_a,
       a.title AS title_a,
       b.sku_id AS sku_b,
       b.merchant AS merchant_b,
       b.title AS title_b,
       round(array_cosine_similarity(a.vector, b.vector), 4) AS similarity,
       round(abs(a.price - b.price) / a.price * 100, 1) AS price_difference_pct
FROM merchant_catalog a
JOIN merchant_catalog b ON a.sku_id < b.sku_id
AND a.merchant != b.merchant
AND array_cosine_similarity(a.vector, b.vector) >= 0.90
AND abs(a.price - b.price) <= (0.20 * a.price)
ORDER BY similarity DESC;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 477.45 ms) 
-- Match identical product listings across merchants (high similarity + price within 20%)

SELECT a.sku_id AS sku_a,
       a.merchant AS merchant_a,
       a.title AS title_a,
       b.sku_id AS sku_b,
       b.merchant AS merchant_b,
       b.title AS title_b,
       round(array_cosine_similarity(a.vector, b.vector), 4) AS similarity,
       round(abs(a.price - b.price) / a.price * 100, 1) AS price_difference_pct
FROM merchant_catalog a
JOIN merchant_catalog b ON a.sku_id < b.sku_id
AND a.merchant != b.merchant
AND array_cosine_similarity(a.vector, b.vector) >= 0.90
AND abs(a.price - b.price) <= (0.20 * a.price)
ORDER BY similarity DESC;
