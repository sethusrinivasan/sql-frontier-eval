Scenario & Frontier Trap
Business Scenario & Problem

4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted

Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 201.63 ms) 
-- Compute triplet margin violation with margin = 0.2
-- Violation = GREATEST(0, dist(a, p) - dist(a, n) + margin)

SELECT triplet_id,
       round(array_distance(anchor_vec, positive_vec), 4) AS dist_pos,
       round(array_distance(anchor_vec, negative_vec), 4) AS dist_neg,
       round(greatest(0.0, array_distance(anchor_vec, positive_vec) - array_distance(anchor_vec, negative_vec) + 0.20), 4) AS triplet_margin_loss,
       CASE
           WHEN (array_distance(anchor_vec, positive_vec) - array_distance(anchor_vec, negative_vec) + 0.20) > 0 THEN 'VIOLATION'
           ELSE 'SATISFIED'
       END AS evaluation_status
FROM eval_triplets
ORDER BY triplet_id;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 198.85 ms) 
-- Compute triplet margin violation with margin = 0.2
 -- Violation = GREATEST(0, dist(a, p) - dist(a, n) + margin)

SELECT triplet_id,
       round(array_distance(anchor_vec, positive_vec), 4) AS dist_pos,
       round(array_distance(anchor_vec, negative_vec), 4) AS dist_neg,
       round(greatest(0.0, array_distance(anchor_vec, positive_vec) - array_distance(anchor_vec, negative_vec) + 0.20), 4) AS triplet_margin_loss,
       CASE
           WHEN (array_distance(anchor_vec, positive_vec) - array_distance(anchor_vec, negative_vec) + 0.20) > 0 THEN 'VIOLATION'
           ELSE 'SATISFIED'
       END AS evaluation_status
FROM eval_triplets
ORDER BY triplet_id;
