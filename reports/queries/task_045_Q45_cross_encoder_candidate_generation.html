-- First-stage candidate extraction: Top candidates in relevant category

SELECT article_id,
       category,
       round(array_cosine_similarity(vec, [0.95, 0.05]::FLOAT[2]), 4) AS stage1_biencoder_score

WHERE category = 'tech'
ORDER BY stage1_biencoder_score DESC
