4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 262.37 ms) 
-- Positive query: [1.0, 0.0] (cyberpunk neon), Negative query: [0.0, 1.0] (blurry/rustic)
-- Net Score = Cosine(vec, pos) - (0.5 * Cosine(vec, neg))

SELECT asset_id,
       name,
       round(array_cosine_similarity(vec, [1.0, 0.0]::FLOAT[2]), 4) AS pos_sim,
       round(array_cosine_similarity(vec, [0.0, 1.0]::FLOAT[2]), 4) AS neg_sim,
       round(array_cosine_similarity(vec, [1.0, 0.0]::FLOAT[2]) - (0.50 * array_cosine_similarity(vec, [0.0, 1.0]::FLOAT[2])), 4) AS net_guided_score
FROM generative_assets
ORDER BY net_guided_score DESC;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 256.42 ms) 
-- Positive query: [1.0, 0.0] (cyberpunk neon), Negative query: [0.0, 1.0] (blurry/rustic)
 -- Net Score = Cosine(vec, pos) - (0.5 * Cosine(vec, neg))

SELECT asset_id,
       name,
       round(array_cosine_similarity(vec, [1.0, 0.0]::FLOAT[2]), 4) AS pos_sim,
       round(array_cosine_similarity(vec, [0.0, 1.0]::FLOAT[2]), 4) AS neg_sim,
       round(array_cosine_similarity(vec, [1.0, 0.0]::FLOAT[2]) - (0.50 * array_cosine_similarity(vec, [0.0, 1.0]::FLOAT[2])), 4) AS net_guided_score
FROM generative_assets
ORDER BY net_guided_score DESC;
