4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 338.75 ms) 
WITH bounds AS
  (SELECT doc_id,
          title,
          raw_bm25,
          vector_sim,
          min(raw_bm25) OVER () AS min_bm25,
          max(raw_bm25) OVER () AS max_bm25,
          min(vector_sim) OVER () AS min_vec,
          max(vector_sim) OVER () AS max_vec
   FROM search_candidates),
     normalized AS
  (SELECT doc_id,
          title,
          coalesce((raw_bm25 - min_bm25) / nullif(max_bm25 - min_bm25, 0), 1.0) AS norm_bm25,
          coalesce((vector_sim - min_vec) / nullif(max_vec - min_vec, 0), 1.0) AS norm_vec
   FROM bounds)
SELECT doc_id,
       title,
       round(norm_bm25, 4) AS norm_bm25,
       round(norm_vec, 4) AS norm_vec,
       round(0.50 * norm_bm25 + 0.50 * norm_vec, 4) AS convex_hybrid_score
FROM normalized
ORDER BY convex_hybrid_score DESC;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 329.95 ms) 
-- Min-Max normalization and 50/50 convex combination
 WITH bounds AS
  (SELECT doc_id,
          title,
          raw_bm25,
          vector_sim,
          min(raw_bm25) OVER () AS min_bm25,
          max(raw_bm25) OVER () AS max_bm25,
          min(vector_sim) OVER () AS min_vec,
          max(vector_sim) OVER () AS max_vec
   FROM search_candidates),
      normalized AS
  (SELECT doc_id,
          title,
          coalesce((raw_bm25 - min_bm25) / nullif(max_bm25 - min_bm25, 0), 1.0) AS norm_bm25,
          coalesce((vector_sim - min_vec) / nullif(max_vec - min_vec, 0), 1.0) AS norm_vec
   FROM bounds)
SELECT doc_id,
       title,
       round(norm_bm25, 4) AS norm_bm25,
       round(norm_vec, 4) AS norm_vec,
       round(0.50 * norm_bm25 + 0.50 * norm_vec, 4) AS convex_hybrid_score
FROM normalized
ORDER BY convex_hybrid_score DESC;
