4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 129.76 ms) 
WITH quartiles AS
  (SELECT quantile_cont(latency_ms, 0.25) AS q1,
          quantile_cont(latency_ms, 0.50) AS median_val,
          quantile_cont(latency_ms, 0.75) AS q3
   FROM response_times)
SELECT r.request_id,
       r.latency_ms,
       round((r.latency_ms - q.median_val) / nullif(q.q3 - q.q1, 0), 3) AS robust_iqr_z_score
FROM response_times r
CROSS JOIN quartiles q
ORDER BY robust_iqr_z_score DESC;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 132.21 ms) 
-- Compute robust IQR-normalized score: (x - Q2) / (Q3 - Q1)
 WITH quartiles AS
  (SELECT quantile_cont(latency_ms, 0.25) AS q1,
          quantile_cont(latency_ms, 0.50) AS median_val,
          quantile_cont(latency_ms, 0.75) AS q3
   FROM response_times)
SELECT r.request_id,
       r.latency_ms,
       round((r.latency_ms - q.median_val) / nullif(q.q3 - q.q1, 0), 3) AS robust_iqr_z_score
FROM response_times r
CROSS JOIN quartiles q
ORDER BY robust_iqr_z_score DESC;
