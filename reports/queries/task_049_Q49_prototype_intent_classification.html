4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 834.05 ms) 
WITH ranked_intents AS
  (SELECT q.query_id,
          q.query_text,
          p.intent_name,
          array_cosine_similarity(q.query_vector, p.prototype_vector) AS intent_similarity,
          row_number() OVER (PARTITION BY q.query_id
                             ORDER BY array_cosine_similarity(q.query_vector, p.prototype_vector) DESC) AS rank_match
   FROM inbound_queries q
   CROSS JOIN intent_prototypes p)
SELECT query_id,
       query_text,
       intent_name AS predicted_intent,
       round(intent_similarity, 4) AS confidence
FROM ranked_intents
WHERE rank_match = 1
ORDER BY query_id;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 834.20 ms) 
-- Classify each query to its closest intent prototype
 WITH ranked_intents AS
  (SELECT q.query_id,
          q.query_text,
          p.intent_name,
          array_cosine_similarity(q.query_vector, p.prototype_vector) AS intent_similarity,
          row_number() OVER (PARTITION BY q.query_id
                             ORDER BY array_cosine_similarity(q.query_vector, p.prototype_vector) DESC) AS rank_match
   FROM inbound_queries q
   CROSS JOIN intent_prototypes p)
SELECT query_id,
       query_text,
       intent_name AS predicted_intent,
       round(intent_similarity, 4) AS confidence
FROM ranked_intents
WHERE rank_match = 1
ORDER BY query_id;
