4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted

Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 240.90 ms) 
-- Joint multi-modal score: 60% image vector similarity + 40% text vector similarity

SELECT item_id,
       title,
       round(array_cosine_similarity(img_vec, [0.82, 0.18]::FLOAT[2]), 4) AS img_sim,
       round(array_cosine_similarity(txt_vec, [0.92, 0.08]::FLOAT[2]), 4) AS txt_sim,
       round(0.60 * array_cosine_similarity(img_vec, [0.82, 0.18]::FLOAT[2]) + 0.40 * array_cosine_similarity(txt_vec, [0.92, 0.08]::FLOAT[2]), 4) AS joint_multimodal_score
FROM fashion_items
ORDER BY joint_multimodal_score DESC;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 231.72 ms) 
-- Joint multi-modal score: 60% image vector similarity + 40% text vector similarity

SELECT item_id,
       title,
       round(array_cosine_similarity(img_vec, [0.82, 0.18]::FLOAT[2]), 4) AS img_sim,
       round(array_cosine_similarity(txt_vec, [0.92, 0.08]::FLOAT[2]), 4) AS txt_sim,
       round(0.60 * array_cosine_similarity(img_vec, [0.82, 0.18]::FLOAT[2]) + 0.40 * array_cosine_similarity(txt_vec, [0.92, 0.08]::FLOAT[2]), 4) AS joint_multimodal_score
FROM fashion_items
ORDER BY joint_multimodal_score DESC;
