4. Queries Executed (Candidate vs Golden Ground Truth) #

Beautified & Syntax Highlighted
Database Engine Context: This query was executed and profiled on DuckDB 2.0 In-Memory Engine with equivalent analytical syntax and execution verification on PostgreSQL 18.7 + Extensions (Docker Compose) and Amazon Aurora Serverless v2 PostgreSQL.
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 76.95 ms) 
WITH basket_totals AS
  (SELECT count(DISTINCT order_id) AS total_baskets
   FROM order_baskets),
     item_counts AS
  (SELECT item,
          count(DISTINCT order_id) AS item_baskets
   FROM order_baskets
   GROUP BY item),
     pair_counts AS
  (SELECT a.item AS item_a,
          b.item AS item_b,
          count(DISTINCT a.order_id) AS pair_baskets
   FROM order_baskets a
   JOIN order_baskets b ON a.order_id = b.order_id
   AND a.item != b.item
   GROUP BY a.item,
            b.item)
SELECT p.item_a,
       p.item_b,
       p.pair_baskets,
       round(p.pair_baskets * 1.0 / t.total_baskets, 3) AS support,
       round(p.pair_baskets * 1.0 / ca.item_baskets, 3) AS confidence,
       round((p.pair_baskets * 1.0 * t.total_baskets) / (ca.item_baskets * cb.item_baskets), 3) AS lift
FROM pair_counts p
JOIN item_counts ca ON p.item_a = ca.item
JOIN item_counts cb ON p.item_b = cb.item
CROSS JOIN basket_totals t
WHERE p.pair_baskets >= 2
ORDER BY lift DESC,
         p.item_a,
         p.item_b;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 76.52 ms) 
-- Compute Support, Confidence, and Lift for Item A -> Item B
 WITH basket_totals AS
  (SELECT count(DISTINCT order_id) AS total_baskets
   FROM order_baskets),
      item_counts AS
  (SELECT item,
          count(DISTINCT order_id) AS item_baskets
   FROM order_baskets
   GROUP BY item),
      pair_counts AS
  (SELECT a.item AS item_a,
          b.item AS item_b,
          count(DISTINCT a.order_id) AS pair_baskets
   FROM order_baskets a
   JOIN order_baskets b ON a.order_id = b.order_id
   AND a.item != b.item
   GROUP BY a.item,
            b.item)
SELECT p.item_a,
       p.item_b,
       p.pair_baskets, -- Support: P(A and B)
 round(p.pair_baskets * 1.0 / t.total_baskets, 3) AS support, -- Confidence: P(B | A) = pair / count_a
 round(p.pair_baskets * 1.0 / ca.item_baskets, 3) AS confidence, -- Lift: Confidence / P(B) = (pair * total) / (count_a * count_b)
 round((p.pair_baskets * 1.0 * t.total_baskets) / (ca.item_baskets * cb.item_baskets), 3) AS lift
FROM pair_counts p
JOIN item_counts ca ON p.item_a = ca.item
JOIN item_counts cb ON p.item_b = cb.item
CROSS JOIN basket_totals t
WHERE p.pair_baskets >= 2
ORDER BY lift DESC,
         p.item_a,
         p.item_b;
