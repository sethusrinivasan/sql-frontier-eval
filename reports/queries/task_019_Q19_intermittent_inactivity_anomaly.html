4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted
Database Engine Context: This query was executed and profiled on DuckDB 2.0 In-Memory Engine with equivalent analytical syntax and execution verification on PostgreSQL 18.7 + Extensions (Docker Compose).
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 323.25 ms) 
WITH purchase_intervals AS
  (SELECT customer_id,
          order_date,
          order_date - lag(order_date) OVER (PARTITION BY customer_id
                                             ORDER BY order_date) AS days_between
   FROM orders_history),
     user_cadence AS
  (SELECT customer_id,
          max(order_date) AS last_order_date,
          median(days_between) AS median_purchase_interval_days
   FROM purchase_intervals
   GROUP BY customer_id)
SELECT customer_id,
       last_order_date,
       median_purchase_interval_days,
       (DATE '2026-03-10' - last_order_date) AS days_since_last_order,
       CASE
           WHEN (DATE '2026-03-10' - last_order_date) > (2.5 * median_purchase_interval_days) THEN 'HIGH_CHURN_RISK'
           ELSE 'NORMAL'
       END AS churn_risk_status
FROM user_cadence
ORDER BY customer_id;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 314.75 ms) 
-- Calculate median inter-purchase days and compare against days elapsed since latest order
 WITH purchase_intervals AS
  (SELECT customer_id,
          order_date,
          order_date - lag(order_date) OVER (PARTITION BY customer_id
                                             ORDER BY order_date) AS days_between
   FROM orders_history),
      user_cadence AS
  (SELECT customer_id,
          max(order_date) AS last_order_date,
          median(days_between) AS median_purchase_interval_days
   FROM purchase_intervals
   GROUP BY customer_id)
SELECT customer_id,
       last_order_date,
       median_purchase_interval_days,
       (DATE '2026-03-10' - last_order_date) AS days_since_last_order,
       CASE
           WHEN (DATE '2026-03-10' - last_order_date) > (2.5 * median_purchase_interval_days) THEN 'HIGH_CHURN_RISK'
           ELSE 'NORMAL'
       END AS churn_risk_status
FROM user_cadence
ORDER BY customer_id;
