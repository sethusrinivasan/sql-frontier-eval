TF-IDF is the foundational algorithm of information retrieval. Calculating it directly in SQL provides deep insight into term uniqueness across a corpus without requiring Python.

4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted
Database Engine Context: This query was executed and profiled on DuckDB 2.0 In-Memory Engine with equivalent analytical syntax and execution verification on PostgreSQL 18.7 + Extensions (Docker Compose).
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 173.64 ms) 
WITH doc_lengths AS
  (SELECT doc_id,
          sum(term_count) AS doc_len
   FROM doc_words
   GROUP BY doc_id),
     corpus_stats AS
  (SELECT count(DISTINCT doc_id) AS total_docs
   FROM doc_words),
     term_docs AS
  (SELECT term,
          count(DISTINCT doc_id) AS docs_with_term
   FROM doc_words
   GROUP BY term)
SELECT w.doc_id,
       w.term,
       w.term_count,
       round(w.term_count * 1.0 / l.doc_len, 4) AS tf,
       round(ln(c.total_docs * 1.0 / td.docs_with_term), 4) AS idf,
       round((w.term_count * 1.0 / l.doc_len) * ln(c.total_docs * 1.0 / td.docs_with_term), 4) AS tfidf_score
FROM doc_words w
JOIN doc_lengths l ON w.doc_id = l.doc_id
JOIN term_docs td ON w.term = td.term
CROSS JOIN corpus_stats c
WHERE w.term = 'columnar'
ORDER BY tfidf_score DESC;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 195.88 ms) 
-- Calculate TF-IDF for term 'columnar' across 3 documents
 -- TF = term_count / total_terms_in_doc, IDF = LN(Total_Docs / Docs_With_Term)
 WITH doc_lengths AS
  (SELECT doc_id,
          sum(term_count) AS doc_len
   FROM doc_words
   GROUP BY doc_id),
      corpus_stats AS
  (SELECT count(DISTINCT doc_id) AS total_docs
   FROM doc_words),
      term_docs AS
  (SELECT term,
          count(DISTINCT doc_id) AS docs_with_term
   FROM doc_words
   GROUP BY term)
SELECT w.doc_id,
       w.term,
       w.term_count,
       round(w.term_count * 1.0 / l.doc_len, 4) AS tf,
       round(ln(c.total_docs * 1.0 / td.docs_with_term), 4) AS idf,
       round((w.term_count * 1.0 / l.doc_len) * ln(c.total_docs * 1.0 / td.docs_with_term), 4) AS tfidf_score
FROM doc_words w
JOIN doc_lengths l ON w.doc_id = l.doc_id
JOIN term_docs td ON w.term = td.term
CROSS JOIN corpus_stats c
WHERE w.term = 'columnar'
ORDER BY tfidf_score DESC;
