Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 195.90 ms) 
-- Query: 'machine' AND 'learning', with a 5.0x boost for exact phrase match

SELECT article_id,
       text_content,
       (CASE
            WHEN text_content ILIKE '%machine%' THEN 1.0
            ELSE 0.0
        END + CASE
                  WHEN text_content ILIKE '%learning%' THEN 1.0
                  ELSE 0.0
              END + CASE
                        WHEN text_content ILIKE '%machine learning%' THEN 5.0
                        ELSE 0.0
                    END) AS relevance_score
FROM articles_proximity
WHERE text_content ILIKE '%machine%'
  AND text_content ILIKE '%learning%'
ORDER BY relevance_score DESC;
