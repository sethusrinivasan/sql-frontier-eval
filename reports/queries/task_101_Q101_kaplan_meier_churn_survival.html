4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 5.10 ms) 
WITH monthly_risk AS
  (SELECT tenure_months,
          count(*) AS cohort_events,
          sum(CASE
                  WHEN has_churned THEN 1
                  ELSE 0
              END) AS churn_count,
          count(*) OVER (
                         ORDER BY tenure_months DESC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS number_at_risk
   FROM subscription_cohort
   GROUP BY tenure_months)
SELECT tenure_months,
       number_at_risk,
       churn_count,
       round(1.0 - (churn_count * 1.0 / number_at_risk), 4) AS interval_survival_rate
FROM monthly_risk
ORDER BY tenure_months;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 8.01 ms) 
-- Kaplan-Meier survival probability estimation by tenure month
 WITH monthly_risk AS
  (SELECT tenure_months,
          count(*) AS cohort_events,
          sum(CASE
                  WHEN has_churned THEN 1
                  ELSE 0
              END) AS churn_count,
          count(*) OVER (
                         ORDER BY tenure_months DESC ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS number_at_risk
   FROM subscription_cohort
   GROUP BY tenure_months)
SELECT tenure_months,
       number_at_risk,
       churn_count,
       round(1.0 - (churn_count * 1.0 / number_at_risk), 4) AS interval_survival_rate
FROM monthly_risk
ORDER BY tenure_months;
