4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 1203.97 ms) 
WITH rolling_data AS
  (SELECT trade_date,
          settlement_amount,
          sum(settlement_amount) OVER w AS window_sum,
          min(settlement_amount) OVER w AS window_min,
          max(settlement_amount) OVER w AS window_max,
          count(*) OVER w AS window_count
   FROM daily_settlement WINDOW w AS (
                                      ORDER BY trade_date RANGE BETWEEN INTERVAL 6 DAY PRECEDING AND CURRENT ROW))
SELECT trade_date,
       settlement_amount,
       CASE
           WHEN window_count > 2 THEN (window_sum - window_min - window_max) / (window_count - 2)
           ELSE NULL
       END AS trimmed_mean_5d
FROM rolling_data
ORDER BY trade_date;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 173.47 ms) 
-- Calculate trimmed rolling mean excluding min and max values within a 5-day window
 WITH windowed_stats AS
  (SELECT trade_date,
          settlement_amount,
          count(*) OVER w AS win_count,
          sum(settlement_amount) OVER w AS win_sum,
          min(settlement_amount) OVER w AS win_min,
          max(settlement_amount) OVER w AS win_max
   FROM daily_settlement WINDOW w AS (
                                      ORDER BY trade_date ROWS BETWEEN 2 PRECEDING AND 2 FOLLOWING))
SELECT trade_date,
       settlement_amount, -- Subtract min and max from sum and divide by (count - 2)
 CASE
     WHEN win_count > 2 THEN round((win_sum - win_min - win_max) / (win_count - 2), 2)
     ELSE settlement_amount
 END AS trimmed_mean_5d
FROM windowed_stats
ORDER BY trade_date;
