4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 147.35 ms) 
WITH eligible_touches AS
  (SELECT c.user_id,
          c.revenue,
          a.channel,
          a.click_time,
          row_number() OVER (PARTITION BY c.user_id
                             ORDER BY a.click_time ASC) AS touch_asc,
          row_number() OVER (PARTITION BY c.user_id
                             ORDER BY a.click_time DESC) AS touch_desc,
          count(*) OVER (PARTITION BY c.user_id) AS total_touches
   FROM conversions c
   JOIN ad_clicks a ON c.user_id = a.user_id
   AND a.click_time >= c.conversion_time - INTERVAL '30 days'
   AND a.click_time <= c.conversion_time)
SELECT channel,
       sum(CASE
               WHEN touch_asc = 1 THEN revenue
               ELSE 0
           END) AS first_touch_revenue,
       sum(CASE
               WHEN touch_desc = 1 THEN revenue
               ELSE 0
           END) AS last_touch_revenue,
       round(sum(revenue / total_touches), 2) AS linear_touch_revenue
FROM eligible_touches
GROUP BY channel
ORDER BY channel;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 157.73 ms) 
-- Multi-touch attribution comparison (First-touch, Last-touch, Linear)
 WITH eligible_touches AS
  (SELECT c.user_id,
          c.revenue,
          a.channel,
          a.click_time,
          row_number() OVER (PARTITION BY c.user_id
                             ORDER BY a.click_time ASC) AS touch_asc,
          row_number() OVER (PARTITION BY c.user_id
                             ORDER BY a.click_time DESC) AS touch_desc,
          count(*) OVER (PARTITION BY c.user_id) AS total_touches
   FROM conversions c
   JOIN ad_clicks a ON c.user_id = a.user_id
   AND a.click_time >= c.conversion_time - INTERVAL '30 days'
   AND a.click_time <= c.conversion_time)
SELECT channel, -- First Touch: 100% credit if first
 sum(CASE
         WHEN touch_asc = 1 THEN revenue
         ELSE 0
     END) AS first_touch_revenue, -- Last Touch: 100% credit if last
 sum(CASE
         WHEN touch_desc = 1 THEN revenue
         ELSE 0
     END) AS last_touch_revenue, -- Linear: Equal credit across all touchpoints
 round(sum(revenue / total_touches), 2) AS linear_touch_revenue
FROM eligible_touches
GROUP BY channel
ORDER BY channel;
