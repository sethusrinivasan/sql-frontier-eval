4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 802.82 ms) 
WITH pairwise_distances AS
  (SELECT a.tx_id,
          b.tx_id AS neighbor_id,
          array_distance(a.vec, b.vec) AS dist,
          row_number() OVER (PARTITION BY a.tx_id
                             ORDER BY array_distance(a.vec, b.vec) ASC) AS k_rank
   FROM transaction_embeddings a
   JOIN transaction_embeddings b ON a.tx_id != b.tx_id),
     knn_avg AS
  (SELECT tx_id,
          avg(dist) AS mean_2nn_distance
   FROM pairwise_distances
   WHERE k_rank <= 2
   GROUP BY tx_id)
SELECT tx_id,
       round(mean_2nn_distance, 4) AS outlier_score,
       CASE
           WHEN mean_2nn_distance > 5.0 THEN 'ANOMALOUS_OUTLIER'
           ELSE 'NORMAL'
       END AS classification
FROM knn_avg
ORDER BY outlier_score DESC;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 821.72 ms) 
-- Calculate average distance to 2 nearest neighbors to score anomaly level
 WITH pairwise_distances AS
  (SELECT a.tx_id,
          b.tx_id AS neighbor_id,
          array_distance(a.vec, b.vec) AS dist,
          row_number() OVER (PARTITION BY a.tx_id
                             ORDER BY array_distance(a.vec, b.vec) ASC) AS k_rank
   FROM transaction_embeddings a
   JOIN transaction_embeddings b ON a.tx_id != b.tx_id),
      knn_avg AS
  (SELECT tx_id,
          avg(dist) AS mean_2nn_distance
   FROM pairwise_distances
   WHERE k_rank <= 2
   GROUP BY tx_id)
SELECT tx_id,
       round(mean_2nn_distance, 4) AS outlier_score,
       CASE
           WHEN mean_2nn_distance > 5.0 THEN 'ANOMALOUS_OUTLIER'
           ELSE 'NORMAL'
       END AS classification
FROM knn_avg
ORDER BY outlier_score DESC;
