Business Scenario & Problem

Models join events without checking timestamp ordering (e.g., verifying after purchase), artificially inflating funnel conversion.

4. Queries Executed (Candidate vs Golden Ground Truth)

Beautified & Syntax Highlighted
Database Engine Context: This query was executed and profiled on DuckDB 2.0 In-Memory Engine with equivalent analytical syntax and execution verification on PostgreSQL 18.7 + Extensions (Docker Compose).
Candidate SQL Executed by gemini-3.8-flash-low (Runtime on 1M Rows: 35.96 ms) 
WITH step1 AS
  (SELECT user_id,
          action_time AS signup_time
   FROM user_actions
   WHERE stage = 'signup'),
     step2 AS
  (SELECT s1.user_id,
          s1.signup_time,
          min(a.action_time) AS verify_time
   FROM step1 s1
   JOIN user_actions a ON s1.user_id = a.user_id
   AND a.stage = 'verify'
   AND a.action_time >= s1.signup_time
   AND a.action_time <= s1.signup_time + INTERVAL '24 hours'
   GROUP BY s1.user_id,
            s1.signup_time),
     step3 AS
  (SELECT s2.user_id,
          s2.signup_time,
          s2.verify_time,
          min(a.action_time) AS purchase_time
   FROM step2 s2
   JOIN user_actions a ON s2.user_id = a.user_id
   AND a.stage = 'purchase'
   AND a.action_time >= s2.verify_time
   AND a.action_time <= s2.verify_time + INTERVAL '24 hours'
   GROUP BY s2.user_id,
            s2.signup_time,
            s2.verify_time)
SELECT count(s1.user_id) AS total_signups,
       count(s2.user_id) AS verified_users,
       count(s3.user_id) AS completed_purchasers,
       round(count(s3.user_id) * 100.0 / count(s1.user_id), 1) AS end_to_end_conversion_pct
FROM step1 s1
LEFT JOIN step2 s2 ON s1.user_id = s2.user_id
LEFT JOIN step3 s3 ON s1.user_id = s3.user_id;
Golden Reference Ground Truth SQL (Runtime on 1M Rows: 31.10 ms) 
-- Enforce Signup -> Verify -> Purchase chronological order
 WITH step1 AS
  (SELECT user_id,
          action_time AS signup_time
   FROM user_actions
   WHERE stage = 'signup'),
      step2 AS
  (SELECT s1.user_id,
          s1.signup_time,
          min(a.action_time) AS verify_time
   FROM step1 s1
   JOIN user_actions a ON s1.user_id = a.user_id
   AND a.stage = 'verify'
   AND a.action_time >= s1.signup_time
   AND a.action_time <= s1.signup_time + INTERVAL '24 hours'
   GROUP BY s1.user_id,
            s1.signup_time),
      step3 AS
  (SELECT s2.user_id,
          s2.signup_time,
          s2.verify_time,
          min(a.action_time) AS purchase_time
   FROM step2 s2
   JOIN user_actions a ON s2.user_id = a.user_id
   AND a.stage = 'purchase'
   AND a.action_time >= s2.verify_time
   AND a.action_time <= s2.verify_time + INTERVAL '24 hours'
   GROUP BY s2.user_id,
            s2.signup_time,
            s2.verify_time)
SELECT count(s1.user_id) AS total_signups,
       count(s2.user_id) AS verified_users,
       count(s3.user_id) AS completed_purchasers,
       round(count(s3.user_id) * 100.0 / count(s1.user_id), 1) AS end_to_end_conversion_pct
FROM step1 s1
LEFT JOIN step2 s2 ON s1.user_id = s2.user_id
LEFT JOIN step3 s3 ON s1.user_id = s3.user_id;
